  2  import json, pathlib

  4  CATALOG = json.load(open(pathlib.Path(__file__).with_name("catalog.json")))
  5  ARCH, BRANDS = CATALOG["archetypes"], set(CATALOG["brands"])

  8  COMPAT = {("audio", "audio"), ("audio", "cv"), ("cv", "cv"), ("cv", "pitch"), ("cv", "audio"),
  9            ("pitch", "pitch"), ("pitch", "cv"), ("gate", "gate"), ("gate", "cv")}

 12  def validate(patch: dict, hp_limit: int = 104) -> dict:
 13      errors, warnings = [], []
 14      mods = patch.get("modules") or []
 15      cables = patch.get("cables") or []
 16      if not isinstance(mods, list) or not mods:
 17          return {"valid": False, "errors": ["patch has no modules"], "warnings": [], "hp_used": 0}
 18  
 19      slots = {}
 20      for m in mods:
 21          slot, arch, brand = m.get("slot"), m.get("archetype"), m.get("brand")
 22          if not slot or not isinstance(slot, str):
 23              errors.append(f"module missing a slot name: {m}"); continue
 24          if slot in slots:
 25              errors.append(f"duplicate slot name '{slot}'"); continue
 26          if arch not in ARCH:
 27              errors.append(f"slot '{slot}': unknown archetype '{arch}' (use list_modules)"); continue
 28          if arch != "OUT" and brand not in BRANDS:
 29              errors.append(f"slot '{slot}': unknown brand '{brand}'")
 30          slots[slot] = ARCH[arch]
 31  
 32      hp_used = sum(a["hp"] for a in slots.values())
 33      if hp_used > hp_limit:
 34          errors.append(f"rack is {hp_used}HP, limit is {hp_limit}HP; drop or shrink something")
 35      outs = [s for s, a in slots.items() if a["role"] == "out"]
 36      if len(outs) != 1:
 37          errors.append(f"patch needs exactly one OUT module, found {len(outs)}")
 38  
 39      def jack(ref):
 40          if not isinstance(ref, str) or "." not in ref:
 41              return None, None, f"bad jack reference '{ref}', want slot.jack"
 42          slot, jid = ref.split(".", 1)
 43          if slot not in slots:
 44              return None, None, f"'{ref}': no slot named '{slot}'"
 45          for j in slots[slot]["jacks"]:
 46              if j["id"] == jid:
 47                  return slot, j, None
 48          have = ", ".join(j["id"] for j in slots[slot]["jacks"])
 49          return slot, None, f"'{ref}': slot '{slot}' has no jack '{jid}' (has: {have})"
 50  
 51      fed = {}
 52      for c in cables:
 53          fs, fj, e1 = jack(c.get("from")); ts, tj, e2 = jack(c.get("to"))
 54          if e1: errors.append(e1)
 55          if e2: errors.append(e2)
 56          if e1 or e2 or fj is None or tj is None:
 57              continue
 58          if fj["dir"] != "out":
 59              errors.append(f"{c['from']} is an input, cables must start at an output")
 60          if tj["dir"] != "in":
 61              errors.append(f"{c['to']} is an output, cables must end at an input")
 62          if fj["dir"] == "out" and tj["dir"] == "in" and (fj["kind"], tj["kind"]) not in COMPAT:
 63              errors.append(f"{c['from']} ({fj['kind']}) into {c['to']} ({tj['kind']}) will not do anything useful")
 64          if c["to"] in fed:
 65              errors.append(f"{c['to']} already has a cable from {fed[c['to']]}; one cable per input")
 66          fed[c["to"]] = c["from"]
 67  
 68      if outs:
 69          o = outs[0]
 70          if not any(k.startswith(o + ".") for k in fed):
 71              errors.append("nothing is patched into OUT; you will hear silence")
 72      for s, a in slots.items():
 73          if a["role"] in ("osc", "voice", "drum", "noise", "fx", "filter", "vca", "mix"):
 74              has_out = any(j["dir"] == "out" for j in a["jacks"])
 75              used = any(f.startswith(s + ".") for f in fed.values())
 76              if has_out and not used and a["role"] != "out":
 77                  warnings.append(f"slot '{s}' ({a['role']}) has no cable leaving it; it is silent")
 78          if a["role"] in ("voice", "drum") and not any(k.startswith(s + ".") for k in fed):
 79              warnings.append(f"slot '{s}' ({a['role']}) has no gate or trigger coming in; it will never fire")
 80  
 81      return {"valid": not errors, "errors": errors, "warnings": warnings, "hp_used": hp_used}




Describe a sound, get back a modular synth patch for Eurorack Grid that has been checked, not just written. The model gets three tools: the catalog by role, the exact jacks on one module, and a validator. The validator is plain Python, the only thing allowed to say valid, and the CLI saves the last patch it accepted, never the model's prose.

A patch is a list of modules and the cables between them, and every jack carries pitch, gate, CV or audio. It is easy for a model to write a patch that looks right and does nothing: a cable into an input that does not exist, audio into a pitch input, two cables on one jack, nothing reaching the output. Every error message tells the model what to do next, because the model is the one reading it.


The catalog is a snapshot of the archetype table, not the live app. Nineteen archetypes share jack arrays defined elsewhere in the app, so for those the jacks were filled in by role and the file marks them jacks_inferred. The validator checks wiring, not sound. A valid patch can still be boring.
